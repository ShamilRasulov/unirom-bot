@dp.callback_query(F.data.startswith("type_"))
async def process_type(callback: CallbackQuery, state: FSMContext):
    type_key = callback.data.replace("type_", "")
    await state.update_data(construction_type=type_key)
    await state.set_state(CalculatorState.waiting_width)
    type_names = {
        "plastic": "Пластиковые окна",
        "aluminum": "Алюминиевые окна",
        "plastic_door": "Пластиковые двери",
        "aluminum_door": "Алюминиевые двери"
    }
    await callback.message.edit_text(f"Выбрано: {type_names.get(type_key, type_key)}\n\nВведите ширину в сантиметрах (например: 120):")
    await callback.answer()

@dp.message(CalculatorState.waiting_width)
async def process_width(message: Message, state: FSMContext):
    try:
        width = float(message.text.replace(",", "."))
        if width <= 0 or width > 500:
            raise ValueError
        await state.update_data(width=width)
        await state.set_state(CalculatorState.waiting_height)
        await message.answer("Введите высоту в сантиметрах (например: 150):")
    except ValueError:
        await message.answer("Введите корректное число (ширина в см, например: 120)")

@dp.message(CalculatorState.waiting_height)
async def process_height(message: Message, state: FSMContext):
    try:
        height = float(message.text.replace(",", "."))
        if height <= 0 or height > 500:
            raise ValueError
        await state.update_data(height=height)
        await state.set_state(CalculatorState.waiting_glass)
        data = await state.get_data()
        width = data["width"]
        area = (width * height) / 10000
        await message.answer(f"Размеры: {width}x{height} см = {area:.2f} м2\n\nВыберите тип стеклопакета:", reply_markup=glass_menu())
    except ValueError:
        await message.answer("Введите корректное число (высота в см, например: 150)")

@dp.callback_query(F.data.startswith("opt_"))
async def process_options(callback: CallbackQuery, state: FSMContext):
    opt = callback.data.replace("opt_", "")
    data = await state.get_data()
    options = data.get("options", [])
    if opt == "done":
        await calculate_final(callback, state)
        return
    if opt in options:
        options.remove(opt)
    else:
        options.append(opt)
    await state.update_data(options=options)
    opt_names = {"mosquito": "Москитная сетка", "sill": "Подоконник", "install": "Монтаж"}
    selected = "\n".join([f"+ {opt_names.get(o, o)}" for o in options]) if options else "Ничего не выбрано"
    await callback.message.edit_text(f"Калькулятор\n\nВыбранные опции:\n{selected}\n\nДобавьте еще или нажмите Готово:", reply_markup=options_menu())
    await callback.answer()

async def calculate_final(callback: CallbackQuery, state: FSMContext):
    data = await state.get_data()
    type_key = data["construction_type"]
    width = data["width"]
    height = data["height"]
    glass = data["glass"]
    options = data.get("options", [])
    price_data = PRICES[type_key]
    area = (width * height) / 10000
    base_cost = price_data["base"] * area
    glass_cost = price_data[glass] * area
    options_cost = 0
    opt_details = []
    opt_names = {
        "mosquito": ("Москитная сетка", price_data["mosquito"]),
        "sill": ("Подоконник", price_data["sill"]),
        "install": ("Монтаж", price_data["install"])
    }
    for opt in options:
        if opt in opt_names and price_data[opt] > 0:
            name, cost = opt_names[opt]
            options_cost += cost
            opt_details.append(f"{name}: {cost:,} сум")
    total = base_cost + glass_cost + options_cost
    result_text = f"""РАСЧЕТ СТОИМОСТИ

Размеры: {width}x{height} см = {area:.2f} м2
Тип: {price_data["name"]}

Детализация:
- Базовая стоимость: {base_cost:,.0f} сум
- Стеклопакет ({glass}): {glass_cost:,.0f} сум
"""
    if opt_details:
        result_text += "\n".join([f"- {d}" for d in opt_details]) + "\n"
    result_text += f"""

ИТОГО: {total:,.0f} сум

Цена ориентировочная. Точная стоимость после замера.

Хотите оставить заявку на замер?"""
    order_kb = InlineKeyboardMarkup(inline_keyboard=[
        [InlineKeyboardButton(text="Оставить заявку", callback_data="order_from_calc")],
        [InlineKeyboardButton(text="Новый расчет", callback_data="calc")],
        [InlineKeyboardButton(text="Главное меню", callback_data="main_menu")]
    ])
    await callback.message.edit_text(result_text, reply_markup=order_kb)
    await callback.answer()
